        const nameInput = document.getElementById('nameInput');

        const userName = document.getElementById('userName');


        // تحديث الاسم فوراً أثناء الكتابة

        nameInput.addEventListener('input', (e) => {

            if(e.target.value.trim() !== "") {

                userName.textContent = e.target.value;

            } else {

                userName.textContent = "اسمك هنا";

            }

        });


        // ميزة ذكية: التقاط الاسم من الرابط تلقائياً إذا أرسل الشخص الرابط مضافاً إليه الاسم

        // مثال: url.com/?name=محمد

        const urlParams = new URLSearchParams(window.location.search);

        const nameParam = urlParams.get('name');

        if (nameParam) {

            userName.textContent = nameParam;

            nameInput.value = nameParam;

        }
